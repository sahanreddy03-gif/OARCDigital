2.2 Speculation Rules API (Next-Level)
Create app/speculation-rules.tsx:
tsx'use client'
import { useEffect } from 'react'

export function SpeculationRules() {
  useEffect(() => {
    const script = document.createElement('script')
    script.type = 'speculationrules'
    script.textContent = JSON.stringify({
      prerender: [
        {
          source: 'list',
          urls: [
            '/services',
            '/case-studies',
            '/contact',
            '/pricing',
            '/about'
          ]
        }
      ],
      prefetch: [
        {
          source: 'document',
          where: {
            and: [
              { href_matches: '/*' },
              { not: { href_matches: '/api/*' } }
            ]
          },
          eagerness: 'moderate'
        }
      ]
    })
    document.head.appendChild(script)
  }, [])
  
  return null
}
Include in app/layout.tsx. Chrome browsers will pre-render pages user is likely to click.

tsimport sharp from 'sharp'
import fs from 'fs'
import path from 'path'

async function generateBlur(imagePath: string) {
  const buffer = await sharp(imagePath)
    .resize(10, 10, { fit: 'inside' })
    .toBuffer()
  return `data:image/jpeg;base64,${buffer.toString('base64')}`
}

3.4 Image Sitemap
Create app/image-sitemap.xml/route.ts:
tsximport { NextResponse } from 'next/server'
import { readdirSync } from 'fs'
import path from 'path'

export async function GET() {
  const imagesDir = path.join(process.cwd(), 'public/images')
  const files = readdirSync(imagesDir)
  
  const xml = `<?xml version="1.0" encoding="UTF-8"?>
<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9"
        xmlns:image="http://www.google.com/schemas/sitemap-image/1.1">
  <url>
    <loc>https://oarcdigital.com/</loc>
    ${files.map(file => `
    <image:image>
      <image:loc>https://oarcdigital.com/images/${file}</image:loc>
      <image:title>${file.replace(/\.[^/.]+$/, '').replace(/-/g, ' ')}</image:title>
    </image:image>`).join('')}
  </url>
</urlset>`
  
  return new NextResponse(xml, {
    headers: { 'Content-Type': 'application/xml' }
  })
}
Submit to Google Search Console. This makes your images findable on Google Image Search = extra traffic.

  })
}
