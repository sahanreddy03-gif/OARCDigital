Neural grid hero — technical brief + starter code

Goal: visually communicate that “AI is active.” Use an animated particle/grid with connecting lines and subtle glows. Use three.js for performance and WebGL. Provide a fallback static video or Lottie for lower-end devices.

Behaviour:
	•	Particles drift slowly; when cursor moves, local attraction forms ephemeral clusters.
	•	Periodic pulses travel across the grid (visual “inference waves”); they should be subtle.
	•	Use brand accent (neon / purple-blue / green ring) for glow.
	•	Fallback: small looping video or canvas animation.

Starter code (three.js — minimal, client-side)

Give this to your front-end dev as neuralGridHero.js. It assumes three.js is available.

// neuralGridHero.js
import * as THREE from 'three';
import { OrbitControls } from 'three/examples/jsm/controls/OrbitControls.js';

export default function initNeuralGrid(container) {
  const w = container.clientWidth;
  const h = Math.max(container.clientHeight, 400);

  const scene = new THREE.Scene();
  const camera = new THREE.PerspectiveCamera(60, w / h, 0.1, 2000);
  camera.position.z = 120;

  const renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true });
  renderer.setPixelRatio(window.devicePixelRatio);
  renderer.setSize(w, h);
  container.appendChild(renderer.domElement);

  // particle system
  const particleCount = Math.min(900, Math.floor((w * h) / 800)); // responsive
  const geometry = new THREE.BufferGeometry();
  const positions = new Float32Array(particleCount * 3);
  const sizes = new Float32Array(particleCount);

  for (let i = 0; i < particleCount; i++) {
    positions[i * 3] = (Math.random() - 0.5) * 300;
    positions[i * 3 + 1] = (Math.random() - 0.5) * 160;
    positions[i * 3 + 2] = (Math.random() - 0.5) * 100;
    sizes[i] = 1 + Math.random() * 2.5;
  }
  geometry.setAttribute('position', new THREE.BufferAttribute(positions, 3));
  geometry.setAttribute('size', new THREE.BufferAttribute(sizes, 1));

  const material = new THREE.ShaderMaterial({
    transparent: true,
    depthTest: false,
    uniforms: {
      uTime: { value: 0.0 },
      uPixelRatio: { value: window.devicePixelRatio },
      uColor: { value: new THREE.Color(0x8efc6b) } // brand neon
    },
    vertexShader: `
      attribute float size;
      uniform float uTime;
      varying float vAlpha;
      void main() {
        vec3 pos = position;
        float t = uTime * 0.0005;
        pos.x += sin(t + position.y * 0.01) * 2.0;
        vec4 mvPosition = modelViewMatrix * vec4( pos, 1.0 );
        gl_PointSize = size * ( 300.0 / -mvPosition.z );
        gl_Position = projectionMatrix * mvPosition;
        vAlpha = 0.5 + 0.5 * sin(uTime * 0.001 + position.x * 0.01);
      }`,
    fragmentShader: `
      uniform vec3 uColor;
      varying float vAlpha;
      void main() {
        float r = length(gl_PointCoord - vec2(0.5));
        float alpha = smoothstep(0.5, 0.0, r) * vAlpha;
        gl_FragColor = vec4(uColor, alpha);
      }`
  });

  const points = new THREE.Points(geometry, material);
  scene.add(points);

  // connecting lines (simple subtle)
  const lineMaterial = new THREE.LineBasicMaterial({ color: 0x204020, transparent: true, opacity: 0.06 });
  const lineGeom = new THREE.BufferGeometry();
  const linePositions = new Float32Array(particleCount * 3);
  lineGeom.setAttribute('position', new THREE.BufferAttribute(linePositions, 3));
  const lines = new THREE.LineSegments(lineGeom, lineMaterial);
  scene.add(lines);

  // mouse interaction
  const mouse = new THREE.Vector2(-1, -1);
  function onMove(e) {
    const rect = renderer.domElement.getBoundingClientRect();
    mouse.x = ((e.clientX - rect.left) / rect.width) * 2 - 1;
    mouse.y = -((e.clientY - rect.top) / rect.height) * 2 + 1;
  }
  window.addEventListener('mousemove', onMove);

  // resize
  window.addEventListener('resize', () => {
    const W = container.clientWidth;
    const H = Math.max(container.clientHeight, 400);
    camera.aspect = W / H;
    camera.updateProjectionMatrix();
    renderer.setSize(W, H);
  });

  // animate
  let last = performance.now();
  function animate(now) {
    const delta = now - last;
    last = now;
    material.uniforms.uTime.value = now;
    // small rotation for depth
    points.rotation.y += delta * 0.00008;
    // update lines positions to subtly follow particles (cheap approach)
    const pos = geometry.attributes.position.array;
    const lp = lineGeom.attributes.position.array;
    for (let i = 0; i < particleCount; i++) {
      lp[i * 3] = pos[i * 3] * 0.98;
      lp[i * 3 + 1] = pos[i * 3 + 1] * 0.98;
      lp[i * 3 + 2] = pos[i * 3 + 2] * 0.98;
    }
    lineGeom.attributes.position.needsUpdate = true;

    renderer.render(scene, camera);
    requestAnimationFrame(animate);
  }
  requestAnimationFrame(animate);

  return {
    dispose: () => {
      window.removeEventListener('mousemove', onMove);
      renderer.dispose();
      container.removeChild(renderer.domElement);
    }
  };
}
