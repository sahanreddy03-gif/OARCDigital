typescriptimport { NextResponse } from 'next/server'

export async function GET() {
  const sitemaps = [
    'sitemap-pages.xml',
    'sitemap-services.xml',
    'sitemap-locations.xml',
    'sitemap-industries.xml',
    'sitemap-blog.xml',
    'sitemap-aeo.xml',
    'sitemap-case-studies.xml',
    'sitemap-ai-agents.xml',
    'sitemap-images.xml',
    'sitemap-videos.xml'
  ]
  
  const xml = `<?xml version="1.0" encoding="UTF-8"?>
<sitemapindex xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">
${sitemaps.map(s => `  <sitemap>
    <loc>https://oarcdigital.com/${s}</loc>
    <lastmod>${new Date().toISOString()}</lastmod>
  </sitemap>`).join('\n')}
</sitemapindex>`
  
  return new NextResponse(xml, { headers: { 'Content-Type': 'application/xml' } })
}
For each sub-sitemap (example for services):
app/sitemap-services.xml/route.ts:
typescriptimport { readdirSync, statSync } from 'fs'
import path from 'path'
import { NextResponse } from 'next/server'

export async function GET() {
  const servicesDir = path.join(process.cwd(), 'app/services')
  const slugs = readdirSync(servicesDir, { withFileTypes: true })
    .filter(d => d.isDirectory())
    .map(d => d.name)
  
  const urls = slugs.map(slug => {
    const filePath = path.join(servicesDir, slug, 'page.tsx')
    const mtime = statSync(filePath).mtime.toISOString()
    return `
  <url>
    <loc>https://oarcdigital.com/services/${slug}</loc>
    <lastmod>${mtime}</lastmod>
    <changefreq>weekly</changefreq>
    <priority>0.9</priority>
  </url>`
  }).join('')
  
  return new NextResponse(
    `<?xml version="1.0" encoding="UTF-8"?>
<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">${urls}
</urlset>`,
    { headers: { 'Content-Type': 'application/xml' } }
  )
}
Do this pattern for all 10 sub-sitemaps.
Real file mtimes = Google trusts the lastmod signal.

Better: Use middleware.ts:
typescriptimport { NextResponse } from 'next/server'

const junkPatterns = [
  /\/old-blog\//,
  /\/legacy\//,
  // patterns from GSC that have zero traffic
]

export function middleware(request) {
  const { pathname } = request.nextUrl
  
  if (junkPatterns.some(p => p.test(pathname))) {
    return new NextResponse(null, { status: 410 })
  }
  
  return NextResponse.next()
}

typescriptconst INDEXNOW_KEY = process.env.INDEXNOW_KEY // Generate at indexnow.org

export async function submitToIndexNow(urls: string[]) {
  await fetch('https://api.indexnow.org/indexnow', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      host: 'oarcdigital.com',
      key: INDEXNOW_KEY,
      keyLocation: `https://oarcdigital.com/${INDEXNOW_KEY}.txt`,
      urlList: urls
    })
  })
}
